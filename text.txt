	btn.Size = UDim2.new(0, 100, 1, 0)

	btn.Font = Enum.Font.GothamMedium

	corner.CornerRadius = UDim.new(0, 6)

	return btn
end

-- Tab switching logic
local function SwitchTab(selectedTab)
	ScriptVaultPage.Visible = (selectedTab == "Vault")
	ExecutorPage.Visible = (selectedTab == "Executor")
	SettingsPage.Visible = (selectedTab == "Settings")

	VaultTabBtn.BackgroundColor3 = (selectedTab == "Vault") and Color3.fromRGB(50, 50, 65) or Color3.fromRGB(35, 35, 45)
	ExecutorTabBtn.BackgroundColor3 = (selectedTab == "Executor") and Color3.fromRGB(50, 50, 65) or Color3.fromRGB(35, 35, 45)
	SettingsTabBtn.BackgroundColor3 = (selectedTab == "Settings") and Color3.fromRGB(50, 50, 65) or Color3.fromRGB(35, 35, 45)
end

VaultTabBtn.MouseButton1Click:Connect(function() SwitchTab("Vault") end)
ExecutorTabBtn.MouseButton1Click:Connect(function() SwitchTab("Executor") end)
SettingsTabBtn.MouseButton1Click:Connect(function() SwitchTab("Settings") end)
SwitchTab("Vault")
